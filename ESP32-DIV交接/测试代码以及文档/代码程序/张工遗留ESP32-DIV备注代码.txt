/* ============================================================================
 * ESP32-DIV —— 主入口文件（ESP32-DIV.ino）
 * ============================================================================
 *
 * 【这是什么工程？】
 *   这是一个基于 ESP32 / ESP32-S3 开发板的“多功能无线渗透测试掌机”，
 *   类 Flipper Zero 的开源替代品。它把屏幕、按键、触摸、各类无线芯片
 *   整合到一块小小的开发板上，做成一个可以随身携带的“黑客瑞士军刀”。
 *
 * 【工程做了什么事？一句话总结】
 *   开机 → 显示 Logo → 进入 2×4 宫格主菜单 →
 *   用户通过物理按键（PCF8574 扩展）或触摸屏选择功能 →
 *   进入对应的子菜单/功能页 → 在使用中按 SELECT 退出返回菜单。
 *
 * 【硬件组成】（对应本文件内使用的库）
 *   1. ESP32 主控                ——  整个设备的大脑（WiFi/BT 一体）
 *   2. TFT 彩色显示屏             ——  TFT_eSPI 库驱动，显示菜单和功能界面
 *   3. PCF8574 I2C 扩展芯片       ——  扩展出 8 个 IO 口外接物理按键
 *      （上/下/左/右/确认 等方向键）
 *   4. 触摸屏（叠加在 TFT 上）     ——  Touchscreen.h 读取触摸坐标，等效于点按按钮
 *   5. NRF24L01+（2.4GHz 模块）   ——  实现 2.4GHz 扫描 / 协议干扰
 *   6. CC1101（SubGHz 模块）      ——  实现 315/433/868/915MHz 重放与干扰
 *   7. PN532（RFID/NFC 读卡模块） ——  实现门禁卡读取/克隆/破解等功能
 *   8. GPS 模块                   ——  记录坐标（配合 WiFi/BT 做 Wardriving）
 *   9. 红外收发管（IR）            ——  录制/回放/万能遥控器
 *  10. SD 卡                      ——  保存配置、固件、抓到的数据包
 *  11. 电池 + 背光 PWM            ——  锂电池供电，PWM 调节屏幕亮度
 *
 * 【菜单结构总览】
 *   主菜单（2 列 × 4 行宫格）
 *   ├─ 0. WiFi               → 8 项子菜单
 *   │     Packet Monitor（抓包）/ Beacon Spammer（SSID洪水）/
 *   │     WiFi Deauther（断连攻击）/ Probe Request Flood（探测泛洪）/
 *   │     Deauth Detector（检测攻击）/ WiFi Scanner（扫描）/ Captive Portal（钓鱼门户）
 *   ├─ 1. 2.4GHz (NRF24L01+) → 3 项子菜单（Scanner 扫描 / Proto Kill 协议杀手）
 *   ├─ 2. More               → 2×2 宫格 → 再进入 IR / RFID / GPS 子菜单
 *   ├─ 3. Settings           → 全屏设置应用（用户配置等）
 *   ├─ 4. Bluetooth (BLE)    → 7 项子菜单（干扰/欺骗/SourApple/嗅探/扫描/键盘注入）
 *   ├─ 5. SubGHz (CC1101)    → 4 项子菜单（重放攻击/干扰/保存的配置）
 *   ├─ 6. Tools              → 5 项子菜单（串口监视器/固件升级/触摸校准/SD文件管理）
 *   └─ 7. About              → 项目介绍页面
 *
 * 【本文件（主函数）在整个工程中的角色】
 *   - 全局菜单状态管理（当前在主菜单还是子菜单、选中了哪一项）
 *   - 屏幕绘制（主菜单宫格 / 子菜单列表 / More 宫格 / About 页）
 *   - 输入分发（物理按键 + 触摸，统一抽象为 up/down/left/right/select）
 *   - 功能启动与退出（选到某个功能 → setup() → 循环 loop() → 按 SELECT 退出）
 *   其余各功能的实际实现分散在 wifi.cpp / bluetooth.cpp / subghz.cpp /
 *   ir.cpp / rfid.cpp / gps.cpp / ducky.cpp / utils.cpp 等文件中。
 *
 * 【操作方式】
 *   物理按键 UP / DOWN / LEFT / RIGHT 移动光标，SELECT 确认进入；
 *   触摸屏直接点格子/列表项即可选中；功能页内触摸底部导航栏、
 *   或按 SELECT 退出返回上一级菜单。
 * ========================================================================== */

/**
 * @brief 设置屏幕背光亮度
 * @param value  0~255 的 PWM 占空比数值（越小越暗，255 最亮）
 *
 * 实现方式：用 ESP32 的 LEDC（硬件 PWM 控制器）输出一个
 * 可调占空比的方波到背光引脚 BACKLIGHT_PIN，占空比越高屏幕越亮。
 * 这样比直接开关背光能实现“无级调光”，省电且护眼。
 */
void setBrightness(uint8_t value) {
  ledcWrite(PWM_CHANNEL, value);  // 把一个 PWM 通道的占空比设为 value
}//设置背光亮度，用1占空比控制

/**
 * @brief 根据当前主菜单项，把“活动子菜单”的指针指向对应的一套菜单数据
 *
 * 作用：updateActiveSubmenu 本质上就是屏幕绘制函数的“万能插头”。
 * 它本身不存任何实际数据，只负责动态地把全局指针 active_submenu_items /
 * active_submenu_size / active_submenu_icons 指向当前子菜单（WiFi、蓝牙或
 * RFID 等）对应的文字数组和图标数组。
 *
 * 好处：当你在菜单间切换时，程序只要把这个“插头”插到对应的套装上，
 * 下方的 displaySubmenu() 绘图函数就能完全不用区分是哪个菜单，
 * 统一用 active_submenu_icons / active_submenu_items 来取数据绘制，
 * 实现了 UI 渲染逻辑的通用化和代码复用。
 */
void updateActiveSubmenu() {
    switch (current_menu_index) {
        case 0:  // 主菜单 [0] WiFi → 指向 WiFi 子菜单
            active_submenu_items = submenu_items;
            active_submenu_size = NUM_SUBMENU_ITEMS;
            active_submenu_icons = wifi_submenu_icons;
            break;
        case 1:  // 主菜单 [1] 2.4GHz → 指向 NRF24L01+ 子菜单
            active_submenu_items = nrf_submenu_items;
            active_submenu_size = nrf_NUM_SUBMENU_ITEMS;
            active_submenu_icons = nrf_submenu_icons;
            break;
        case 2:  // 主菜单 [2] More → 根据 other_layer 指向 IR / RFID / GPS / 首页
            if (other_layer == OTHER_LAYER_HOME) {
                active_submenu_items = other_submenu_items;
                active_submenu_size = other_NUM_SUBMENU_ITEMS;
                active_submenu_icons = other_submenu_icons;
            } else if (other_layer == OTHER_LAYER_IR) {
                active_submenu_items = ir_submenu_items;
                active_submenu_size = ir_NUM_SUBMENU_ITEMS;
                active_submenu_icons = ir_submenu_icons;
            } else if (other_layer == OTHER_LAYER_RFID) {
                active_submenu_items = rfid_submenu_items;
                active_submenu_size = rfid_NUM_SUBMENU_ITEMS;
                active_submenu_icons = rfid_submenu_icons;
            } else if (other_layer == OTHER_LAYER_GPS) {
                active_submenu_items = gps_submenu_items;
                active_submenu_size = gps_NUM_SUBMENU_ITEMS;
                active_submenu_icons = gps_submenu_icons;
            } else {
                // 兜底：意外值也回退到 More 首页
                active_submenu_items = other_submenu_items;
                active_submenu_size = other_NUM_SUBMENU_ITEMS;
                active_submenu_icons = other_submenu_icons;
            }
            break;
        case 3:  // 主菜单 [3] Settings → 全屏设置页，不用子菜单列表
            active_submenu_items = nullptr;
            active_submenu_size = 0;
            active_submenu_icons = nullptr;
            break;
        case 4:  // 主菜单 [4] Bluetooth → 指向 BLE 子菜单
            active_submenu_items = bluetooth_submenu_items;
            active_submenu_size = bluetooth_NUM_SUBMENU_ITEMS;
            active_submenu_icons = bluetooth_submenu_icons;
            break;
        case 5:  // 主菜单 [5] SubGHz → 指向 SubGHz 子菜单
            active_submenu_items = subghz_submenu_items;
            active_submenu_size = subghz_NUM_SUBMENU_ITEMS;
            active_submenu_icons = subghz_submenu_icons;
            break;
        case 6:  // 主菜单 [6] Tools → 指向 Tools 子菜单
            active_submenu_items = tools_submenu_items;
            active_submenu_size = tools_NUM_SUBMENU_ITEMS;
            active_submenu_icons = tools_submenu_icons;
            break;
        case 7:  // 主菜单 [7] About → 全屏页面，不用子菜单列表
            active_submenu_items = nullptr;
            active_submenu_size = 0;
            active_submenu_icons = nullptr;
            break;

        default:
            // 未知主菜单索引：全部置空，避免越界访问
            active_submenu_items = nullptr;
            active_submenu_size = 0;
            active_submenu_icons = nullptr;
            break;
    }
}

// 说明：在部分功能页（如 Tools、RFID、GPS 等）屏幕底部会显示一排
//       5 个可触摸的“虚拟按键”（◀ ▼ ✕ ▲ ▶），等效于物理按键。
//       这一整套机制由下面的状态变量和函数实现。
static bool touchButtonInputEnabled = false;
//决定底部那排触摸按钮是否启用。只有在 Tools 或 RFID 等需要触摸输入的功能里，它才会被设为 true，平时是关着的，防止你在主菜单误触底部。
static bool touchButtonCueDrawn = false;
//记录这排按钮已经画在屏幕上了没有。防止程序重复绘制导致屏幕闪烁或资源浪费，画过一次就锁住。
static bool s_touchNavLabelsConfigured = false;
//记录你有没有给这五个按钮起过自定义名字（比如把中间的“确认”改成“开始”）。用于判断该画默认图标还是你指定的文字。
static bool s_touchNavHeld[5] = {false, false, false, false, false};
//一个长度为 5 的数组，分别记录五个按钮刚才是不是被按着的。用来消除重复触发——你按下不放，它只会触发一次“点击”动作，防止因为手指停留而反复执行命令。
#if HAS_PCF8574_BUTTONS
static bool s_pcfButtonLastState[8] = {true, true, true, true, true, true, true, true};//按键初始化：记录 8 个物理按键上一次的电平，用于检测“按下沿（Edge）”
#endif
static FeatureUI::Button s_touchNavBtns[5];
static const char* s_touchNavLabels[5] = {nullptr, nullptr, nullptr, nullptr, nullptr};
static constexpr int16_t TOUCH_NAV_BAR_H = (FeatureUI::FOOTER_H * 4) / 5;  // 触摸导航栏的高度  标准的80%

/**
 * @brief 把“触摸槽位索引”转换为对应的物理按键宏（BTN_XXX）
 * @param idx  0~4：触摸栏 5 个槽位（从左到右）
 * @return 对应的按键宏（BTN_LEFT/DOWN/SELECT/UP/RIGHT），无效索引返回 -1
 *
 * 作用：触摸驱动告诉你“用户点了槽位 2”，这个函数就返回 BTN_SELECT，
 *      这样程序就能像处理物理按键一样处理这次触摸。
 */
static int touchNavPinForIndex(int idx) {
  switch (idx) {
    case 0: return BTN_LEFT;
    case 1: return BTN_DOWN;
    case 2: return BTN_SELECT;// 中间槽位 = 确认/返回键
    case 3: return BTN_UP;
    case 4: return BTN_RIGHT;
    default: return -1;//在 C/C++ 编程中，-1 是公认的“错误/无效”标志。 如果 idx 是 5、6、100 或者其他任何数，执行这里
  }
}

/**
 * @brief 启用/禁用底部触摸导航栏
 * @param enabled  true=显示并启用触摸按键；false=隐藏并清理相关状态
 *
 * 启用时会立刻把导航栏画出来；禁用时会清掉自定义文字和按住状态，
 * 避免下次打开还残留旧的按钮文字。
 */
void setTouchButtonInputEnabled(bool enabled) {// ← enabled 是传进来的参数
  if (touchButtonInputEnabled != enabled) {  // ← 判断“当前状态”和“目标状态”是否不同
    touchButtonCueDrawn = false;
    if (!enabled) {  // 要关闭时，清理所有临时状态
      for (int i = 0; i < 5; ++i) {
        s_touchNavLabels[i] = nullptr;   // 清除自定义文字
      }
      s_touchNavLabelsConfigured = false;
      for (int i = 0; i < 5; ++i) {
        s_touchNavHeld[i] = false;       // 清除按住记忆
      }
    }
  }
  touchButtonInputEnabled = enabled;// ← 把状态更新为目标值
#if TOUCH_BUTTON_CUE_ENABLED//条件编译：只有编译开关打开时才真正绘制
  if (enabled && feature_active) {
    drawTouchNavBar();
    touchButtonCueDrawn = true;
  }
#endif
}

/** @brief 计算 5 个触摸按钮在屏幕上的矩形坐标（均分宽度，贴屏幕底部） */
static void layoutTouchNavBtns() {
  const int barY = tft.height() - TOUCH_NAV_BAR_H;  // 导航栏顶部 Y
  const int barH = TOUCH_NAV_BAR_H;                 // 导航栏高度
  const int totalW = tft.width();                   // 屏幕总宽
  const int cellW = totalW / 5;                     // 每个按钮等分宽度

  for (int i = 0; i < 5; ++i) {
    const int x = i * cellW;
    const int w = (i == 4) ? (totalW - x) : cellW;  // 最右按钮补足剩余宽度
    s_touchNavBtns[i] = {
      (int16_t)x, (int16_t)barY, (int16_t)w, (int16_t)barH,
      nullptr, FeatureUI::ButtonStyle::Secondary, false};
  }
}

/** @brief 把过长的按钮文字裁剪成 "...xxx..." 形式，保证能放进按钮宽度内 */
static String fitTouchNavLabel(const char* label, int maxWidth) {
  if (!label || !label[0]) {//label[0] 等价于 *label；空字符串直接返回空
    return String();
  }
  String out = label;// 复制一份原始文字
  if (tft.textWidth(out) <= maxWidth) {  // 本来就放得下就直接返回
    return out;
  }
  // 逐字去掉末尾，直到“加省略号”后能放下为止
  while (out.length() > 1 && tft.textWidth(out + "...") > maxWidth) {
    out.remove(out.length() - 1);//remove,length都是String类的方法：删掉最后一个字符
  }
  return out + "...";//把裁剪后的 out 和 "..." 组合成一个完整的显示字符串返回。
}

/** @brief 绘制底部触摸导航栏（背景 + 分隔线 + 5 个按钮的图标/文字） */
static void drawTouchNavBar() {
  static const unsigned char* kIcons[5] = {  // 默认 5 个按钮图标
    bitmap_icon_LEFT,
    bitmap_icon_DOWN,
    bitmap_icon_go_back,   // 中间 = 返回/确认
    bitmap_icon_UP,
    bitmap_icon_RIGHT,
  };
  constexpr int kIconSize = 16;

  const int barY = tft.height() - TOUCH_NAV_BAR_H;
  const int barH = TOUCH_NAV_BAR_H;
  const int barW = tft.width();

  layoutTouchNavBtns();  // 先算好每个按钮的位置

  tft.fillRect(0, barY, barW, barH, UI_FG);        // 画背景条
  tft.drawFastHLine(0, barY, barW, UI_LINE);       // 画顶部横线

  for (int i = 0; i < 5; ++i) {
    const auto& b = s_touchNavBtns[i];
    if (i > 0) {
      tft.drawFastVLine(b.x, barY + 3, barH - 6, UI_LINE);  // 按钮之间画竖分隔线
    }

    if (s_touchNavLabels[i] && s_touchNavLabels[i][0]) {   // 有自定义文字 → 画文字
      tft.setTextDatum(MC_DATUM);                          // 居中对齐
      const uint16_t txtColor = (i == 2) ? UI_ICON : UI_TEXT;
      tft.setTextColor(txtColor, UI_FG);
      const String fit = fitTouchNavLabel(s_touchNavLabels[i], b.w - 8);
      tft.drawString(fit, b.x + b.w / 2, b.y + b.h / 2, 1);
    } else {  // 没有自定义文字 → 画默认箭头图标
      const int ix = b.x + (b.w - kIconSize) / 2;
      const int iy = b.y + (b.h - kIconSize) / 2;
      // 如果配置过自定义标签但某个槽位是 nullptr → 用灰色圆点表示“禁用位”
      const bool inactiveSlot = s_touchNavLabelsConfigured && !s_touchNavLabels[i];
      const unsigned char* icon = inactiveSlot ? bitmap_icon_dots : kIcons[i];
      const uint16_t iconColor = inactiveSlot ? LIGHT_GRAY : ((i == 2) ? UI_ICON : UI_TEXT);
      tft.drawBitmap(ix, iy, icon, kIconSize, kIconSize, iconColor);
    }
  }

  tft.setTextFont(1);
  tft.setTextSize(1);
  tft.setTextColor(UI_TEXT, FEATURE_BG);
}

// ---------------------------------------------------------------------------
// 按键抽象层：把【物理按键】和【触摸导航栏虚拟按键】统一起来
// ---------------------------------------------------------------------------
// 说明：所有菜单代码只调用 isButtonPressed(BTN_XXX) / isButtonPressedEdge(BTN_XXX)，
//      函数内部自动先查物理按键（PCF8574），再查屏幕底部触摸导航栏，
//      对上层代码完全透明——无论是按实体键还是点屏幕，效果一样。

/** @brief 反向查找：哪个触摸槽位对应给定的物理按键宏（BTN_XXX） */
static int touchNavIndexForPin(int buttonPin) {
  for (int i = 0; i < 5; ++i) {
    if (touchNavPinForIndex(i) == buttonPin) {  // 槽位 i 映射到该按键？
      return i;
    }
  }
  return -1;  // 找不到 → 无效槽位
}

/** @brief 检查某个触摸槽位此刻是否被手指按住（且手指落在导航栏区域） */
static bool isTouchNavSlotDown(int idx) {
  if (idx < 0 || !feature_active || !touchButtonInputEnabled) {
    return false;  // 未启用触摸导航，或索引无效 → 肯定没按
  }

  int x = 0;
  int y = 0;
  if (!readTouchXYDismiss(x, y)) {
    return false;  // 当前没有触摸事件（readTouchXYDismiss 会顺便消抖）
  }

  layoutTouchNavBtns();

  const int stripTop = tft.height() - TOUCH_NAV_BAR_H;
  if (y < stripTop) {
    return false;  // 手指点在了导航栏上方的功能内容区，不算导航按钮
  }

  return FeatureUI::hit(s_touchNavBtns, 5, x, y) == idx;  // 命中哪个槽位
}

/** @brief 读取物理按键（PCF8574 扩展 IO）：按下返回 true（低电平有效） */
bool isPhysicalButtonPressed(int buttonPin) {
#if HAS_PCF8574_BUTTONS
  if (getPcf8574Address() != 0) {       // 只有探测到 PCF8574 芯片才读
    return !pcf.digitalRead(buttonPin); // 按键按下 = 该引脚被拉低 = !1 → true
  }
#endif
  return false;  // 未启用物理按键 → 恒为 false
}

/** @brief 触摸导航栏上是否按下了对应“键”（把触摸映射成按键） */
bool isTouchNavButtonPressed(int buttonPin) {
  const int idx = touchNavIndexForPin(buttonPin);
  if (idx < 0) {
    return false;
  }
  return isTouchNavSlotDown(idx);
}

/** @brief 统一按键查询：物理按键 OR 触摸导航栏（任一被按都视为按下） */
bool isButtonPressed(int buttonPin) {
  if (isPhysicalButtonPressed(buttonPin)) {
    return true;
  }
  return isTouchNavButtonPressed(buttonPin);
}

/** @brief 触摸导航按键的“按下沿”检测：只在按下那一刻返回 true 一次 */
bool isTouchNavButtonPressedEdge(int buttonPin) {
  if (!feature_active || !touchButtonInputEnabled) {
    return false;
  }

  const int navIdx = touchNavIndexForPin(buttonPin);
  if (navIdx < 0) {
    return false;
  }

  const bool down = isTouchNavSlotDown(navIdx);
  const bool edge = down && !s_touchNavHeld[navIdx];  // 之前是松开、现在是按下 → 边沿
  s_touchNavHeld[navIdx] = down;                       // 记录最新状态
  return edge;
}

/** @brief 统一“按下沿”检测：物理按键（去抖） + 触摸导航栏 */
bool isButtonPressedEdge(int buttonPin) {
#if HAS_PCF8574_BUTTONS
  if (getPcf8574Address() != 0) {
    const int idx = buttonPin % 8;                 // 只保留 0~7 的数据位
    const bool cur = pcf.digitalRead(buttonPin);
    const bool edge = !cur && s_pcfButtonLastState[idx];  // 低电平且上一轮是高 → 刚按下
    s_pcfButtonLastState[idx] = cur;              // 更新上一轮电平
    if (edge) {
      return true;
    }
  }
#endif

  return isTouchNavButtonPressedEdge(buttonPin);
}

/** @brief 退出功能的统一判定：按 SELECT（物理键或触摸中间键）就算退出 */
bool featureExitButtonPressed() {
  return isPhysicalButtonPressed(BTN_SELECT) || isTouchNavButtonPressed(BTN_SELECT);
}

/** @brief 功能不可用时弹通知提示（如 BLE Ducky 需要 S3 芯片） */
static void showFeatureUnavailable(const char* featureName, const char* requirement) {
  feature_active = false;
  feature_exit_requested = false;
  showNotification(featureName, requirement);  // 屏幕顶部弹提示条
  delay(250);
}

/** @brief 运行“BLE Rubber Ducky”（蓝牙键盘注入）功能 */
static void runBleDuckyFeature() {
#if FEATURE_BLE_DUCKY   // 只有 ESP32-S3 编译开关打开时才编译
  current_submenu_index = 5;
  in_sub_menu = true;
  feature_active = true;
  feature_exit_requested = false;
  Ducky::enter();       // 初始化蓝牙键盘（HID）并准备注入脚本
  while (current_submenu_index == 5 && !feature_exit_requested) {
      current_submenu_index = 5;
      in_sub_menu = true;
      Ducky::loop();    // 循环执行 Ducky 脚本中的按键序列
  }

  Ducky::exit();        // 结束后断开蓝牙、清理状态
  if (feature_exit_requested) {
      in_sub_menu = true;
      is_main_menu = false;
      submenu_initialized = false;
      feature_active = false;
      feature_exit_requested = false;
      displaySubmenu();  // 回到蓝牙子菜单列表
      delay(200);
  }
#else
  showFeatureUnavailable("BLE Rubber Ducky", "This feature requires ESP32-S3.");
#endif
}

/**

 *
 * 作用：使用 updateActiveSubmenu() 设好的 active_submenu_items / active_submenu_icons，
 *      在屏幕上画出一列菜单项（图标 + 文字）。只重画“旧高亮行”和“新高亮行”，
 *      避免每次按键都刷新整个屏幕，减少闪烁。
 *
 * 特殊处理：如果处于“More 首页”（主菜单[2]且 other_layer==HOME），
 *      直接转交给 displayOtherMenuGrid() 画 2×2 宫格。
 */
void displaySubmenu() {
    setTouchButtonInputEnabled(false);  // 子菜单列表页不需要底部触摸导航栏

    if (current_menu_index == 2 && other_layer == OTHER_LAYER_HOME) {
        displayOtherMenuGrid();  // More 首页用宫格而非列表
        return;
    }

    menu_initialized = false;
    last_menu_index = -1;

    tft.setTextFont(2);
    tft.setTextSize(1);

    if (!submenu_initialized) {   // 需要全量重画整个子菜单列表
        tft.fillScreen(UI_BG);

        for (int i = 0; i < active_submenu_size; i++) {   // 逐项绘制 图标+文字
            int yPos = 30 + i * 30;                      // 每行高度 30 像素
            if (i == active_submenu_size - 1) yPos += 10; // 最后一行（返回）多留 10 像素

            tft.setTextColor((i == active_submenu_size - 1) ? UI_TEXT : UI_TEXT, UI_BG);
            tft.drawBitmap(10, yPos, active_submenu_icons[i], 16, 16, (i == active_submenu_size - 1) ? UI_TEXT : UI_TEXT);
            tft.setCursor(30, yPos);
            if (i < active_submenu_size - 1) {
                tft.print("| ");
            }
            tft.print(active_submenu_items[i]);
        }

        submenu_initialized = true;
        last_submenu_index = -1;
    }

    if (last_submenu_index != current_submenu_index) {
        if (last_submenu_index >= 0) {
            int prev_yPos = 30 + last_submenu_index * 30;
            if (last_submenu_index == active_submenu_size - 1) prev_yPos += 10;

            tft.setTextColor((last_submenu_index == active_submenu_size - 1) ? UI_TEXT : UI_TEXT, UI_BG);
            tft.drawBitmap(10, prev_yPos, active_submenu_icons[last_submenu_index], 16, 16, (last_submenu_index == active_submenu_size - 1) ? UI_TEXT : UI_TEXT);
            tft.setCursor(30, prev_yPos);
            if (last_submenu_index < active_submenu_size - 1) {
                tft.print("| ");
            }
            tft.print(active_submenu_items[last_submenu_index]);
        }

        int new_yPos = 30 + current_submenu_index * 30;
        if (current_submenu_index == active_submenu_size - 1) new_yPos += 10;

        tft.setTextColor((current_submenu_index == active_submenu_size - 1) ? UI_ICON : UI_ICON, UI_BG);
        tft.drawBitmap(10, new_yPos, active_submenu_icons[current_submenu_index], 16, 16, (current_submenu_index == active_submenu_size - 1) ? UI_ICON : UI_ICON);
        tft.setCursor(30, new_yPos);
        if (current_submenu_index < active_submenu_size - 1) {
            tft.print("| ");
        }
        tft.print(active_submenu_items[current_submenu_index]);

        last_submenu_index = current_submenu_index;
    }

    drawStatusBar(currentBatteryVoltage, true);
}

// Tools 子菜单各项的索引常量（与 tools_submenu_items 下标一一对应）
constexpr int TOOLS_IDX_TERMINAL = 0;    // [0] 串口监视器
constexpr int TOOLS_IDX_UPDATE   = 1;    // [1] 固件更新
constexpr int TOOLS_IDX_TOUCH    = 2;    // [2] 触摸校准
constexpr int TOOLS_IDX_SD_FILES = 3;    // [3] SD 文件管理器
constexpr int TOOLS_IDX_SETTINGS = -1;   // 保留（未用于设置）
constexpr int TOOLS_IDX_BACK     = 4;    // [4] 返回主菜单

/** @brief 工具功能退出后的统一清理：关闭触摸导航、重置标签、重绘子菜单 */
static void runToolsFeatureExitCleanup() {
    in_sub_menu = true;
    is_main_menu = false;
    submenu_initialized = false;
    feature_active = false;
    feature_exit_requested = false;
    setTouchButtonInputEnabled(false);
    setTouchNavLabels(nullptr, nullptr, nullptr, nullptr, nullptr);
    resetTouchNavHeldState();
    displaySubmenu();
    delay(200);
    while (isButtonPressed(BTN_SELECT)) {
    }
}

/**
 * @brief 通用“工具功能启动器”：任何 Tools 子功能都走这套流程
 * @param idx       子菜单下标（用于 while 循环退出判断）
 * @param setupFn   该功能的初始化函数指针（如 Terminal::terminalSetup）
 * @param loopFn    该功能的主循环函数指针（如 Terminal::terminalLoop）
 *
 * 流程：启用触摸导航（触摸校准除外）→ 调用 setupFn() → 循环调用 loopFn()
 *      直到用户请求退出 → 统一清理（runToolsFeatureExitCleanup）。
 */
static void runToolsFeature(int idx, void (*setupFn)(), void (*loopFn)()) {
    const bool useTouchNav = (idx != TOOLS_IDX_TOUCH);
    current_submenu_index = idx;
    in_sub_menu = true;
    feature_active = true;
    feature_exit_requested = false;
    if (useTouchNav) {
        setTouchButtonInputEnabled(true);
    }
    setupFn();
    while (current_submenu_index == idx && !feature_exit_requested) {
        current_submenu_index = idx;
        in_sub_menu = true;
        loopFn();
        if (feature_exit_requested) {
            break;
        }
        if (!useTouchNav && isButtonPressed(BTN_SELECT)) {
            break;
        }
    }
    runToolsFeatureExitCleanup();
}

    switch (idx) {
        case TOOLS_IDX_TERMINAL:

            break;
        case TOOLS_IDX_TOUCH:
            runToolsFeature(idx, TouchCalib::setup, TouchCalib::loop);
            break;
        case TOOLS_IDX_SD_FILES:
            runToolsFeature(idx, SdFileManager::setup, SdFileManager::loop);
            break;
        default:
            break;
    }
}

/**
 * @brief 处理 Tools 工具子菜单的按键/触摸导航
 *
 * 负责 Tools 子菜单（5 项）的交互：
 *   - UP/DOWN 移动光标；SELECT 启动工具（或返回主菜单）
 *   - 触摸屏直接点某项等同按 SELECT
 *   - 具体工具通过 launchToolsFeature() 统一启动
 */
void handleToolsSubmenuButtons() {
    if (isButtonPressed(BTN_UP)) {
        current_submenu_index = (current_submenu_index - 1 + active_submenu_size) % active_submenu_size;
        last_interaction_time = millis();
        displaySubmenu();
        delay(200);
    }

    if (isButtonPressed(BTN_DOWN)) {
        current_submenu_index = (current_submenu_index + 1) % active_submenu_size;
        last_interaction_time = millis();
        displaySubmenu();
        delay(200);
    }

    if (isButtonPressed(BTN_SELECT)) {
        last_interaction_time = millis();
        delay(200);

        if (current_submenu_index == TOOLS_IDX_BACK) {
            in_sub_menu = false;
            feature_active = false;
            feature_exit_requested = false;
            displayMenu();
            handleButtons();
            is_main_menu = false;
            return;
        }

        launchToolsFeature(current_submenu_index);
        return;
    }

    if (!feature_active) {
        int x, y;
        if (!readTouchXY(x, y)) {
            return;
        }

        for (int i = 0; i < active_submenu_size; i++) {
            int yPos = 30 + i * 30;
            if (i == active_submenu_size - 1) yPos += 10;

            int button_x1 = 10;
            int button_y1 = yPos;
            int button_x2 = 110;
            int button_y2 = yPos + (i == active_submenu_size - 1 ? 40 : 30);

            if (x >= button_x1 && x <= button_x2 && y >= button_y1 && y <= button_y2) {
                current_submenu_index = i;
                last_interaction_time = millis();
                displaySubmenu();
                delay(200);

                if (current_submenu_index == TOOLS_IDX_BACK) {
                    in_sub_menu = false;
                    feature_active = false;
                    feature_exit_requested = false;
                    displayMenu();
                    handleButtons();
                    is_main_menu = false;
                } else {
                    launchToolsFeature(current_submenu_index);
                }
                break;
            }
        }
    }
}

/**
 * @brief 等待用户按下 SELECT/LEFT 或触摸屏幕来关掉“功能不可用”弹窗
 *
 * 用于 RFID 等硬件未检测到的情况：先显示 notification 提示，
 * 然后在此等待用户操作确认，最后刷新回子菜单列表。
 */
static void otherDismissPlaceholder() {
    delay(25);
    while (isButtonPressed(BTN_SELECT) || isButtonPressed(BTN_LEFT)) {
        delay(5);
    }
    while (!isButtonPressed(BTN_SELECT) && !isButtonPressed(BTN_LEFT)) {
        int x = 0, y = 0;
        if (!readTouchXYDismiss(x, y) && !readTouchXY(x, y)) {
            delay(12);
            continue;
        }
        if (isNotificationVisible()) {
            NotificationAction a = notificationHandleTouch(x, y);
            if (a != NotificationAction::None) {
                break;
            }
            hideNotification();
        }
        break;
    }
    if (in_sub_menu) {
        submenu_initialized = false;
        if (current_menu_index == 2 && other_layer == OTHER_LAYER_HOME) {
            other_menu_grid_initialized = false;
            last_other_menu_index = -1;
        }
        displaySubmenu();
    }
}

/**
 * @brief RFID/GPS 功能返回前的“按键释放守卫”
 *
 * 作用是等用户松开所有按键/触摸后再返回菜单，防止刚退出功能时
 * 残留的按键事件立即误触发菜单里的下一个动作（抖动过滤）。
 */
static void otherRfidReturnGuard() {
    delay(120);
    for (int i = 0; i < 120; i++) {
        if (!isButtonPressed(BTN_SELECT) && !isButtonPressed(BTN_LEFT) &&
            !isButtonPressed(BTN_RIGHT) && !isButtonPressed(BTN_UP) &&
            !isButtonPressed(BTN_DOWN) && !isTouchDownDismiss()) {
            break;
        }
        delay(5);
    }
    delay(120);
}

/**
 * @brief 启动某个 RFID/NFC 子功能（PN532 模块）
 * @param idx  RFID 子菜单下标（0~7），对应对应 session 函数
 *
 * 先检查 RfidNfc::begin() 是否成功；失败则弹窗提示并等待关闭。
 * 成功后循环执行对应的 sessionXXX()，直到用户退出或会话结束，
 * 最后恢复 SPI 总线（SD 卡与 PN532 共用 SPI，需归还）并刷新菜单。
 */
static void otherRfidPlaceholderAction(int idx) {
    feature_active = true;
    if (!RfidNfc::begin()) {
        showNotification("RFID/NFC", "PN532 not found. Check SPI wiring/pins.");
        otherDismissPlaceholder();
        feature_active = false;
        return;
    }
    feature_exit_requested = false;
    setTouchButtonInputEnabled(true);
    for (;;) {
        RfidNfc::clearSessionRetry();
        switch (idx) {
            case 0:
                RfidNfc::sessionCardReader();
                break;
            case 1:
                RfidNfc::sessionClone();
                break;
            case 2:
                RfidNfc::sessionErase();
                break;
            case 3:
                RfidNfc::sessionDump();
                break;
            case 4:
                RfidNfc::sessionDecodeAccess();
                break;
            case 5:
                RfidNfc::sessionJamReader();
                break;
            case 6:
                RfidNfc::sessionTagDisrupt();
                break;
            case 7:
                RfidNfc::sessionDisruptEmulate();
                break;
            default:
                feature_active = false;
                restoreSdAfterSharedSpi();
                return;
        }
        if (feature_exit_requested || !RfidNfc::consumeSessionRetry()) {
            break;
        }
        feature_exit_requested = false;
    }
    restoreSdAfterSharedSpi();
    otherRfidReturnGuard();
    submenu_initialized = false;
    displaySubmenu();
    feature_active = false;
}

/**
 * @brief 启动某个 GPS 子功能
 * @param idx  GPS 子菜单下标：0=Wardriver（持续记录坐标+扫描），1=卫星扫描
 *
 * Wardriver 会循环调用 GpsWardriver::session() 直到用户退出；
 * 卫星扫描只跑一次（GpsSatelliteScanner::session()）。
 */
static void otherGpsPlaceholderAction(int idx) {
    feature_active = true;
    if (idx == 0) {
        feature_exit_requested = false;
        setTouchButtonInputEnabled(true);
        for (;;) {
            GpsWardriver::clearSessionRetry();
            GpsWardriver::session();
            if (feature_exit_requested || !GpsWardriver::consumeSessionRetry()) {
                break;
            }
            feature_exit_requested = false;
        }
        setTouchButtonInputEnabled(false);
    } else {
        switch (idx) {
            case 1:
                GpsSatelliteScanner::session();
                break;
            default:
                feature_active = false;
                return;
        }
    }
    otherRfidReturnGuard();
    submenu_initialized = false;
    displaySubmenu();
    feature_active = false;
}

 * 清屏后逐行打印项目名称、开发者、版本、联系邮箱、GitHub、官网
 * （用 tftPrintlnObf 输出“混淆字符串”，避免源码里明文暴露作者信息）。
 * 按 SELECT/LEFT 或触摸屏幕任意处返回主菜单。
 */
void handleAboutPage() {

  feature_active = true;
  feature_exit_requested = false;

  tft.fillScreen(UI_BG);
  currentBatteryVoltage = readBatteryVoltage();
  drawStatusBar(currentBatteryVoltage, true);

  tft.setTextColor(UI_TEXT, UI_BG);
  tft.setTextSize(1);
  tft.setTextFont(2);

  const char* title = "[About This Project]";
  tft.setCursor(10, 50);
  tft.println(title);

  int lineHeight = 18;
  int text_x = 10;
  int text_y = 80;
  tft.setCursor(text_x, text_y);
  tft.print("- ");
  tftPrintlnObf(OBF_PN, sizeof(OBF_PN));
  text_y += lineHeight;
  tft.setCursor(text_x, text_y);
  tft.print("- Developed by: ");
  tftPrintlnObf(OBF_DN, sizeof(OBF_DN));
  text_y += lineHeight;
  tft.setCursor(text_x, text_y);
  tft.print("- Version: ");
  tft.println(ESP32DIV_VERSION);
  text_y += lineHeight;
  tft.setCursor(text_x, text_y);
  tft.print("- Contact: ");
  tftPrintlnObf(OBF_EM, sizeof(OBF_EM));
  text_y += lineHeight;
  tft.setCursor(text_x, text_y);
  tft.print("- GitHub: ");
  tftPrintlnObf(OBF_GH, sizeof(OBF_GH));
  text_y += lineHeight;
  tft.setCursor(text_x, text_y);
  tft.print("- Website: ");
  tftPrintlnObf(OBF_WB, sizeof(OBF_WB));

  tft.setTextFont(1);
  tft.setCursor(10, 300);
  tft.println("Press SELECT / touch to go back");

  while (!feature_exit_requested) {
    if (isButtonPressed(BTN_SELECT) || isButtonPressed(BTN_LEFT)) {
      last_interaction_time = millis();
      feature_exit_requested = true;
      delay(200);
      break;
    }

    int x, y;
    if (readTouchXY(x, y)) {
      last_interaction_time = millis();
      feature_exit_requested = true;
      delay(200);
      break;
    }

    delay(20);
  }

  feature_active = false;
  feature_exit_requested = false;
  in_sub_menu = false;
  submenu_initialized = false;

  menu_initialized = false;
  last_menu_index = -1;
  is_main_menu = false;
  displayMenu();
}

/**
 * @brief 显示“设置”全屏页面
 *
 * 直接调用 AppSettingsUI::setup() 初始化设置界面，然后不断调用
 * AppSettingsUI::loop() 直到用户要求退出（feature_exit_requested），
 * 最后重绘主菜单。
 */
void handleSettingsSubmenuButtons() {

  feature_active = true;
  feature_exit_requested = false;

  AppSettingsUI::setup();
  while (!feature_exit_requested) {
    AppSettingsUI::loop();
  }

  feature_active = false;
  feature_exit_requested = false;

  in_sub_menu = false;
  submenu_initialized = false;

  menu_initialized = false;
  last_menu_index = -1;
  is_main_menu = false;
  displayMenu();
}

/**
 * @brief 全局输入分发：把按键/触摸事件路由到对应的处理器
 *
 * 如果当前在子菜单（in_sub_menu==true）：
 *   根据 current_menu_index 路由到各子菜单处理器
 *   （0=WiFi、1=NRF、2=More、4=BLE、5=SubGHz、6=Tools；
 *    3=Settings 和 7=About 使用全屏页不走这里）。
 * 否则（主菜单）：
 *   处理 UP/DOWN/LEFT/RIGHT 四方向移动光标（2×4 宫格环绕移动），
 *   SELECT 进入对应子菜单（Settings/About 直接打开全屏页），
 *   以及触摸屏点按方块的高亮反馈与确认。
 */
void handleButtons() {
    if (in_sub_menu) {
        switch (current_menu_index) {

            case 0: handleWiFiSubmenuButtons(); break;
            case 1: handleNRFSubmenuButtons(); break;
            case 2: handleOtherSubmenuButtons(); break;
            case 3: /* Settings: full-screen AppSettings, not list submenu */ break;
            case 4: handleBluetoothSubmenuButtons(); break;
            case 5: handleSubGHzSubmenuButtons(); break;
            case 6: handleToolsSubmenuButtons(); break;
            default: break;
        }
    } else {

        if (isButtonPressed(BTN_UP) && !is_main_menu) {
            current_menu_index--;
            if (current_menu_index < 0) {
                current_menu_index = NUM_MENU_ITEMS - 1;
            }
            last_interaction_time = millis();
            displayMenu();
            delay(200);
        }

        if (isButtonPressed(BTN_DOWN) && !is_main_menu) {
            current_menu_index++;
            if (current_menu_index >= NUM_MENU_ITEMS) {
                current_menu_index = 0;
            }
            last_interaction_time = millis();
            displayMenu();
            delay(200);
        }

        if (isButtonPressed(BTN_LEFT) && !is_main_menu) {
            int row = current_menu_index % 4;
            if (current_menu_index >= 4) {
                current_menu_index = row;
            } else if (current_menu_index == 0) {
                current_menu_index = 3;
            } else {
                current_menu_index = row - 1;
            }
            last_interaction_time = millis();
            displayMenu();
            delay(200);
        }

        if (isButtonPressed(BTN_RIGHT) && !is_main_menu) {
            int row = current_menu_index % 4;
            if (current_menu_index < 4) {
                current_menu_index = row + 4;
            } else if (current_menu_index == 7) {
                current_menu_index = 0;
            } else {
                current_menu_index = row + 5;
            }
            last_interaction_time = millis();
            displayMenu();
            delay(200);
        }

        if (isButtonPressed(BTN_SELECT)) {
            last_interaction_time = millis();
            delay(200);

            if (current_menu_index == 3) {
                handleSettingsSubmenuButtons();
            } else if (current_menu_index == 7) {
                handleAboutPage();
            } else {
                updateActiveSubmenu();

                if (active_submenu_items && active_submenu_size > 0) {
                    current_submenu_index = 0;
                    if (current_menu_index == 2) {
                        other_layer = OTHER_LAYER_HOME;
                        other_menu_grid_initialized = false;
                        last_other_menu_index = -1;
                    }
                    in_sub_menu = true;
                    submenu_initialized = false;
                    displaySubmenu();
                }

                if (is_main_menu) {
                    is_main_menu = false;
                    displayMenu();
                } else {
                    is_main_menu = true;
                }
            }
        }

        static unsigned long lastTouchTime = 0;
        const unsigned long touchFeedbackDelay = 100;

        if (!feature_active && (millis() - lastTouchTime >= touchFeedbackDelay)) {
            int x, y;
            if (!readTouchXY(x, y)) { return; }
            delay(10);
        for (int i = 0; i < NUM_MENU_ITEMS; i++) {
                int column = i / 4;
                int row = i % 4;
                int x_position = (column == 0) ? X_OFFSET_LEFT : X_OFFSET_RIGHT;
                int y_position = Y_START + row * Y_SPACING;

                int button_x1 = x_position;
                int button_y1 = y_position;
                int button_x2 = x_position + 100;
                int button_y2 = y_position + 60;

                if (x >= button_x1 && x <= button_x2 && y >= button_y1 && y <= button_y2) {
                    current_menu_index = i;
                    last_interaction_time = millis();
                    displayMenu();

                    unsigned long startTime = millis();
                    while (isTouchDownDismiss() && (millis() - startTime < touchFeedbackDelay)) {
                        delay(10);
                    }

                    if (isTouchDownDismiss()) {

                        if (current_menu_index == 3) {
                            handleSettingsSubmenuButtons();
                        } else if (current_menu_index == 7) {
                            handleAboutPage();
                        } else {
                            updateActiveSubmenu();

                            if (active_submenu_items && active_submenu_size > 0) {
                                current_submenu_index = 0;
                                if (current_menu_index == 2) {
                                    other_layer = OTHER_LAYER_HOME;
                                    other_menu_grid_initialized = false;
                                    last_other_menu_index = -1;
                                }
                                in_sub_menu = true;
                                submenu_initialized = false;
                                displaySubmenu();
                            } else {
                                if (is_main_menu) {
                                    is_main_menu = false;
                                    displayMenu();
                                } else {
                                    is_main_menu = true;
                                }
                            }
                        }
                    }
                    delay(200);
                    break;
                }
            }
        }
    }
}

/**
 * @brief 主程序初始化（Arduino 开机只执行一次）
 *
 * 执行顺序：
 *   1. 串口 115200 初始化（调试日志输出）
 *   2. TFT 屏幕初始化与旋转（TFT_ROTATION）
 *   3. 背光 PWM 初始化并全亮（setBrightness(100)）
 *   4. 应用主题配色 → 黑屏 → 画加载动画（loading()）→ 画 Logo
 *   5. 初始化 SD 卡（initSDCard）→ 加载用户设置（settingsLoad）
 *      → 按设置应用主题和背光亮度
 *   6. 初始化 PCF8574 物理按键（若当前板型启用 HAS_PCF8574_BUTTONS）
 *   7. 初始化 BLE 协议栈（BLEDevice::init）和 BLE 橡皮鸭（若启用 FEATURE_BLE_DUCKY）
 *   8. 启动后台任务：WiFi 后台扫描、BLE 后台扫描、状态栏刷新任务
 *   9. 绘制主菜单 + 状态栏（含电池电压）
 *   10. 初始化触摸屏（setupTouchscreen）
 *   11. 记录起始交互时间，串口输出 "[boot] ready"
 */
void setup() {

  Serial.begin(115200);
  Serial.println("[boot] start");

  tft.init();
  tft.setRotation(TFT_ROTATION);

  ledcSetup(PWM_CHANNEL, PWM_FREQ, PWM_RESOLUTION);
  ledcAttachPin(BACKLIGHT_PIN, PWM_CHANNEL);
  setBrightness(100);

  applyThemeToPalette(settings().theme);

  tft.fillScreen(TFT_BLACK);

  loading(100, UI_ICON, 0, 0, 2, true);

  tft.fillScreen(TFT_BLACK);

  displayLogo(TFT_WHITE, 500);

  initSDCard();

  settingsLoad();
  applyThemeToPalette(settings().theme);
  setBrightness(settings().brightness);

#if HAS_PCF8574_BUTTONS
  if (!initPcf8574Buttons()) {
    Serial.println("PCF8574 buttons unavailable");
  }
#else
  Serial.println("PCF8574 buttons disabled for this board");
#endif

  Serial.println("[boot] BLE init");
  BLEDevice::init(ESP32DIV_NAME);

#if FEATURE_BLE_DUCKY
  Ducky::setup();
#endif

  Serial.println("[boot] background tasks");
  WifiScan::startBackgroundScanner();
  BleScan::startBackgroundScanner();
  startStatusBarTask();

  Serial.println("[boot] draw menu");
  menu_initialized = false;
  currentBatteryVoltage = readBatteryVoltage();
  displayMenu();
  drawStatusBar(currentBatteryVoltage, false);

  setupTouchscreen();

  last_interaction_time = millis();
  Serial.println("[boot] ready");
}

/**
 * @brief 主循环（Arduino 反复执行）
 *
 * 每轮做三件事：
 *   1. 应用当前主题配色（applyThemeToPalette）——用户切换主题后即时生效
 *   2. 处理按键/触摸输入（handleButtons）
 *   3. 刷新顶部状态栏（updateStatusBar，由后台任务驱动显示电池/时间等）
 */
void loop() {
  applyThemeToPalette(settings().theme);
  handleButtons();
  updateStatusBar();
}
